document.getElementById('form').addEventListener('submit', async function(e) { e.preventDefault(); const nome = document.getElementById('nome').value.trim(); const email = document.getElementById('email').value.trim(); const botao = document.getElementById('btnPagar'); // Validação if (!nome || nome.length < 3) { alert('Digite seu nome completo!'); return; } if (!email.includes('@') || !email.includes('.')) { alert('Digite um e-mail válido!'); return; } // Desabilita botão botao.disabled = true; botao.innerHTML = '⏳ Gerando PIX...'; // Pega motorista da URL const params = new URLSearchParams(window.location.search); const motorista = params.get('motorista') || 'LARA'; try { // Chama N8N const webhookUrl = 'https://prazermauricio.app.n8n.cloud/webhook/Pagar'; const response = await fetch(`${webhookUrl}?nome=${encodeURIComponent(nome)}&email=${encodeURIComponent(email)}&motorista=${encodeURIComponent(motorista)}`); const data = await response.json(); if (data.sucesso && data.pixCopiaECola) { const pixCode = data.pixCopiaECola; // Copia o PIX automaticamente try { await navigator.clipboard.writeText(pixCode); } catch (err) { console.log('Erro ao copiar:', err); } // Detecta se é mobile const isMobile = /iPhone|iPad|iPod|Android/i.test(navigator.userAgent); if (isMobile) { // Tenta abrir o app do banco (deep link universal) window.location.href = `https://www.asaas.com/i/${data.cobrancaId}`; // Mostra instruções após 2 segundos setTimeout(() => { alert('✅ PIX copiado!\n\n📱 Clique em "Abrir Pix" na página que abriu.\n\n💡 Se não abrir automaticamente, cole o código PIX no seu banco.\n\n📧 Após pagar, você receberá o livro por email em até 5 minutos.'); }, 2000); } else { // Desktop window.open(data.invoiceUrl, '_blank'); alert('✅ PIX copiado!\n\n📱 Acesse a página que abriu, clique em "Abrir Pix" ou cole o código no seu banco.\n\n📧 Após pagar, você receberá o livro por email.'); } } else { throw new Error('Erro ao gerar PIX'); } } catch (error) { console.error('Erro:', error); alert('❌ Erro ao gerar pagamento. Tente novamente.'); } finally { botao.disabled = false; botao.innerHTML = '💳 Gerar PIX Agora'; } });
